```javascript /* ========================================================= ITEM CARD - حساب الرصيد الفعلي حسب اتجاه الحركة ========================================================= */ function renderItemCard(){ const itemId = document.getElementById("cardItem")?.value || ""; const warehouseId = document.getElementById("cardWarehouse")?.value || ""; const from = document.getElementById("cardFrom")?.value || ""; const to = document.getElementById("cardTo")?.value || ""; const body = document.getElementById("itemCardBody"); const totalQtyEl = document.getElementById("itemCardTotalQty"); const totalValueEl = document.getElementById("itemCardTotalValue"); if(!body) return; /* ===================================================== لا يوجد صنف ===================================================== */ if(!itemId){ body.innerHTML = ` اختر الصنف `; if(totalQtyEl) totalQtyEl.textContent = qty(0); if(totalValueEl) totalValueEl.textContent = money(0); return; } /* ===================================================== تحديد هل المستخدم اختار كل المخازن ===================================================== */ const allWarehouses = !warehouseId || warehouseId === "all" || warehouseId === "ALL"; /* ===================================================== جلب حركات الصنف ===================================================== */ let movements = db.movements .filter(m => String(m.product_id) === String(itemId) ) .filter(m => { /* إذا تم اختيار مخزن/فرع محدد: نعرض فقط الحركات الخاصة بهذا المخزن. */ if(!allWarehouses){ if( String(m.warehouse_id) !== String(warehouseId) ){ return false; } } const d = dateOnly(m.movement_date); if(from && d < from) return false; if(to && d > to) return false; return true; }) .slice() .sort((a,b) => { const dateCompare = String(a.movement_date) .localeCompare( String(b.movement_date) ); if(dateCompare !== 0) return dateCompare; return String( a.created_at || "" ).localeCompare( String( b.created_at || "" ) ); }); /* ===================================================== دالة كمية الحركة ===================================================== */ function getMovementQty(m){ return Number( m.quantity ?? m.qty ?? 0 ); } /* ===================================================== دالة قيمة الحركة ===================================================== */ function getMovementValue(m){ const value = Number( m.value ?? m.total_cost ?? 0 ); if(value !== 0) return value; const q = getMovementQty(m); const cost = Number( m.cost ?? m.unit_cost ?? 0 ); return q * cost; } /* ===================================================== تحديد تأثير الحركة على الرصيد + = زيادة - = نقص ===================================================== */ function getMovementEffect(m){ const q = getMovementQty(m); if(!(q > 0)) return 0; const type = String( m.movement_type || "" ).toLowerCase(); switch(type){ /* رصيد أول المدة */ case "opening": return q; /* شراء / استلام */ case "purchase": return q; /* إذن استلام */ case "receipt": return q; case "receiving": return q; case "in": return q; /* تحويل وارد */ case "transfer_in": return q; /* تسوية زيادة */ case "adjustment_in": return q; /* إذن صرف */ case "issue": return -q; case "withdrawal": return -q; case "out": return -q; /* تحويل صادر */ case "transfer_out": return -q; /* تسوية نقص */ case "adjustment_out": return -q; /* أي حركة غير معروفة */ default: return 0; } } /* ===================================================== حساب الكميات والقيم ===================================================== */ let totalQty = 0; let totalValue = 0; /* ===================================================== في حالة مخزن محدد ===================================================== */ if(!allWarehouses){ movements.forEach(m => { const effect = getMovementEffect(m); if(effect === 0) return; const value = getMovementValue(m); /* القيمة أيضًا تأخذ نفس اتجاه الحركة */ totalQty += effect; totalValue += effect > 0 ? Math.abs(value) : -Math.abs(value); }); } /* ===================================================== في حالة "كل المخازن والفروع" مهم جدًا: التحويل الداخلي يجب ألا يزيد إجمالي الشركة. transfer_out = -4 transfer_in = +4 النتيجة = 0 ===================================================== */ else{ movements.forEach(m => { const type = String( m.movement_type || "" ).toLowerCase(); const q = getMovementQty(m); const value = getMovementValue(m); if(!(q > 0)) return; /* التحويلات الداخلية: لا نضيفها ولا ننقصها من إجمالي مخزون الشركة. */ if( type === "transfer_in" || type === "transfer_out" ){ return; } const effect = getMovementEffect(m); if(effect === 0) return; totalQty += effect; totalValue += effect > 0 ? Math.abs(value) : -Math.abs(value); }); } /* ===================================================== منع ظهور قيم سالبة صغيرة بسبب الكسور العشرية ===================================================== */ if( Math.abs(totalQty) < 0.000001 ){ totalQty = 0; } if( Math.abs(totalValue) < 0.000001 ){ totalValue = 0; } /* ===================================================== عرض الإجمالي ===================================================== */ if(totalQtyEl) totalQtyEl.textContent = qty(totalQty); if(totalValueEl) totalValueEl.textContent = money(totalValue); /* ===================================================== عرض الحركات ===================================================== */ const rows = movements .map(m => { const type = String( m.movement_type || "" ).toLowerCase(); const movementQty = getMovementQty(m); const movementValue = getMovementValue(m); /* تحديد اتجاه الحركة للعرض */ let sign = ""; let displayQty = movementQty; if( type === "transfer_out" || type === "adjustment_out" || type === "issue" || type === "withdrawal" || type === "out" ){ sign = "-"; } else if( type === "transfer_in" || type === "adjustment_in" || type === "opening" || type === "purchase" || type === "receipt" || type === "receiving" || type === "in" ){ sign = "+"; } /* في كارت الصنف: نظهر الكمية مع الإشارة */ const quantityDisplay = sign + qty(displayQty); return ` ${esc( dateOnly( m.movement_date ) )} ${esc( m.document_no || "-" )} ${movementTypeName( m.movement_type )} ${esc( m.from || "-" )} ${esc( m.to || "-" )} ${quantityDisplay} ${money( Number( m.cost ?? m.unit_cost ?? 0 ) )} ${money( movementValue )} `; }) .join(""); /* ===================================================== لا توجد حركات ===================================================== */ body.innerHTML = rows || ` لا توجد حركات `; } ```
